// Spike A: single-frame decoder behaviour on degraded synthetic EAN-13 frames.
import fs from 'node:fs';
import { renderFrame, randomEan13, mulberry32, toPng } from './ean13.mjs';
import { zxingDecode, zbarDecode, repoDecode } from './decoders.mjs';

const W = 1280, H = 720, camera = { focalPx: 1100 };
const N = Number(process.env.N || 16);
const rng = mulberry32(20260903);
const EANS = Array.from({ length: N }, () => randomEan13(rng));
const at = (distanceMm, extra = {}) => { const { pose, ...rest } = extra; return { pose: { distanceMm, cx: 640, cy: 360, ...(pose || {}) }, ...rest }; };
const barPx = (d) => (95 * 0.33) * camera.focalPx / d; // bar-area width in px at distance d
const cutRight = (d, v) => W + barPx(d) / 2 - barPx(d) * v; // cx so that fraction v of the bar area (from its left) is inside
const cutLeft = (d, v) => barPx(d) * v - barPx(d) / 2;

const CONDITIONS = [
  ['dist 120mm (≈3.0 px/module)', at(120)],
  ['dist 180mm (≈2.0 px/module)', at(180)],
  ['dist 240mm (≈1.5 px/module)', at(240)],
  ['dist 300mm (≈1.2 px/module)', at(300)],
  ['dist 360mm (≈1.0 px/module)', at(360)],
  ['dist 420mm (≈0.86 px/module)', at(420)],
  ['small code 80% mag @240mm (≈1.2 px/module)', at(240, { xDimMm: 0.264 })],
  ['blur σ0.7 @180mm', at(180, { blurSigma: 0.7 })],
  ['blur σ1.2 @180mm', at(180, { blurSigma: 1.2 })],
  ['blur σ1.8 @180mm', at(180, { blurSigma: 1.8 })],
  ['blur σ1.0 @240mm', at(240, { blurSigma: 1.0 })],
  ['motion 4px @180mm', at(180, { motionBlurPx: 4 })],
  ['motion 7px @180mm', at(180, { motionBlurPx: 7 })],
  ['motion 11px @180mm', at(180, { motionBlurPx: 11 })],
  ['roll 20° @180mm', at(180, { pose: { rollDeg: 20 } })],
  ['roll 45° @180mm', at(180, { pose: { rollDeg: 45 } })],
  ['roll 80° @180mm', at(180, { pose: { rollDeg: 80 } })],
  ['yaw 30° @180mm', at(180, { pose: { yawDeg: 30 } })],
  ['yaw 45° @180mm', at(180, { pose: { yawDeg: 45 } })],
  ['yaw 60° @180mm', at(180, { pose: { yawDeg: 60 } })],
  ['pitch 45° @180mm', at(180, { pose: { pitchDeg: 45 } })],
  ['can r=33mm yaw 20° @180mm', at(180, { curvatureRadiusMm: 33, pose: { yawDeg: 20 } })],
  ['bottle r=20mm yaw 25° @180mm', at(180, { curvatureRadiusMm: 20, pose: { yawDeg: 25 } })],
  ['glare on code @180mm', at(180, { glare: { x: 660, y: 350, r: 60, strength: 1.0 } })],
  ['low contrast 0.35 + noise 6 @180mm', at(180, { contrast: 0.35, noiseSigma: 6 })],
  ['noise σ15 @180mm', at(180, { noiseSigma: 15 })],
  ['noise σ25 @180mm', at(180, { noiseSigma: 25 })],
  ['blur σ1.0 + noise 8 + yaw 25° @200mm', at(200, { blurSigma: 1.0, noiseSigma: 8, pose: { yawDeg: 25 } })],
  ['cut: only left 55% visible @180mm', at(180, { pose: { cx: cutRight(180, 0.55) } })],
  ['cut: only left 80% visible @180mm', at(180, { pose: { cx: cutRight(180, 0.80) } })],
  ['cut: only right 55% visible @180mm', at(180, { pose: { cx: cutLeft(180, 0.55) } })],
  ['cut: right quiet zone missing (100% bars) @180mm', at(180, { pose: { cx: cutRight(180, 1.0) } })],
  ['damaged: 5px light scratch across code @180mm', at(180, { defect: 'scratch' })],
  ['damaged: 14px dark blot on code @180mm', at(180, { defect: 'blot' })],
  ['two codes side by side @180mm', at(180, { second: true })],
];

function applyDefect(f, kind) {
  const { gray, width, quad } = f;
  const cx = (quad[0][0] + quad[1][0]) / 2, top = quad[0][1], bot = quad[3][1];
  if (kind === 'scratch') { const x = Math.round(cx - 22); for (let y = Math.round(top) - 2; y < Math.round(bot) + 2; y++) for (let dx = 0; dx < 5; dx++) gray[y * width + x + dx] = 230; }
  if (kind === 'blot') { const x = Math.round(cx + 18), y = Math.round((top + bot) / 2); for (let yy = y - 7; yy < y + 7; yy++) for (let xx = x - 7; xx < x + 7; xx++) gray[yy * width + xx] = 30; }
}

const rows = [];
let example = 0;
for (const [name, cond] of CONDITIONS) {
  const stat = { zxing: [0, 0, 0, 0], zxfast: [0, 0, 0, 0], zbar: [0, 0, 0, 0], repo: [0, 0, 0, 0] }; // ok, wrong, ms, errorResults
  for (const ean of EANS) {
    const { defect, second, ...rest } = cond;
    const f = renderFrame({ ean, width: W, height: H, camera, ...rest, supersample: 3, blurSigma: 0.5 + (rest.blurSigma || 0) });
    if (defect) applyDefect(f, defect);
    let gray = f.gray;
    if (second) { const g2 = renderFrame({ ean: EANS[(EANS.indexOf(ean) + 1) % N], width: W, height: H, camera, supersample: 2, pose: { distanceMm: 180, cx: 640 + 260, cy: 360 }, blurSigma: 0.5 }).gray; for (let i = 0; i < gray.length; i++) gray[i] = Math.min(gray[i], g2[i]); }
    if (example < CONDITIONS.length && ean === EANS[0]) fs.writeFileSync(`out/a_${String(example).padStart(2, '0')}_${name.replace(/[^a-z0-9]+/gi, '_')}.png`, toPng(gray, W, H)), example++;
    const truth = second ? new Set([ean, EANS[(EANS.indexOf(ean) + 1) % N]]) : new Set([ean]);
    const runs = {
      zxing: await zxingDecode(gray, W, H),
      zxfast: await zxingDecode(gray, W, H, { tryHarder: false, tryDownscale: false }),
      zbar: await zbarDecode(gray, W, H),
      repo: await repoDecode(gray, W, H),
    };
    for (const [k, r] of Object.entries(runs)) {
      const valid = r.results.filter((x) => x.isValid !== false && x.text && /^\d{13}$/.test(x.text));
      const hit = valid.some((x) => truth.has(x.text));
      const wrong = valid.some((x) => !truth.has(x.text));
      stat[k][0] += hit ? 1 : 0; stat[k][1] += wrong ? 1 : 0; stat[k][2] += r.ms;
      stat[k][3] += r.results.filter((x) => x.isValid === false).length > 0 ? 1 : 0;
    }
  }
  const fmt = (s) => `${Math.round((100 * s[0]) / N)}%${s[1] ? ` (✗${s[1]})` : ''} ${(s[2] / N).toFixed(0)}ms`;
  rows.push(`| ${name} | ${fmt(stat.zxing)} | ${fmt(stat.zxfast)} | ${fmt(stat.zbar)} | ${fmt(stat.repo)} | ${stat.zxing[3]}/${N} |`);
  console.log(rows[rows.length - 1]);
}
const md = `## Spike A — single-frame decoders on synthetic EAN-13 frames (1280×720, focal 1100 px ≈ 60° HFOV; baseline lens blur σ0.5 px on every frame, 3× supersampling; N=${N} random EANs per condition; Node ${process.version} on Apple Silicon, single thread — phone CPU is ~3–6× slower)\n\n| condition | zxing-wasm tryHarder | zxing-wasm fast | zbar-wasm | repo scanline | zxing error-results (detect w/o decode) |\n|---|---|---|---|---|---|\n${rows.join('\n')}\n\n(✗n = frames where a decoder returned a WRONG 13-digit value; error-results = frames where zxing returned a symbol with a checksum/decode error but with a position)\n`;
fs.writeFileSync('out/spike_a_results.md', md);
console.log('\nwritten out/spike_a_results.md');
